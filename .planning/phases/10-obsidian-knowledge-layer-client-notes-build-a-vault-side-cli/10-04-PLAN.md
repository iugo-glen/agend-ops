    ```python
    def _extract_managed_section(source: Path, section_name: str) -> str | None:
        text = source.read_text(encoding="utf-8")
        start_re = re.compile(rf"<!--\s*{re.escape(section_name)}-START\s*-->")
        end_re = re.compile(rf"<!--\s*{re.escape(section_name)}-END\s*-->")
        s = start_re.search(text)
        e = end_re.search(text)
        if not s or not e or s.end() > e.start():
            print(f"WARN: source {source} missing or malformed {section_name} markers; skipping section",
                  file=sys.stderr)
            return None
        return text[s.end():e.start()].strip("\n")
    ```

    2. **`_handle_icloud_placeholder(target: Path) -> bool`** — if `<target>.icloud` placeholder file exists alongside target (or replacing target), call `brctl download <parent dir>` with retries. Return True on success, False after all retries fail.

    ```python
    import subprocess
    import time

    def _handle_icloud_placeholder(target: Path, max_retries: int = 3, delay_s: float = 2.0) -> bool:
        """If target is a .icloud placeholder, attempt brctl download. Returns True if downloaded."""
        placeholder = target.parent / f".{target.name}.icloud"
        if not placeholder.exists() and target.exists():
            return True  # No placeholder, file present — nothing to do
        if not placeholder.exists():
            return True  # Neither placeholder nor target — caller will create new
        for attempt in range(max_retries):
            try:
                subprocess.run(
                    ["brctl", "download", str(target.parent)],
                    check=True, capture_output=True, timeout=30,
                )
                # Wait for the file to materialize
                time.sleep(delay_s)
                if target.exists() and not placeholder.exists():
                    return True
            except (subprocess.SubprocessError, FileNotFoundError):
                pass  # brctl not available or failed — try again
            time.sleep(delay_s)
        return False
    ```

    3. **`_update_frontmatter_last_synced(target: Path, last_synced_iso: str) -> None`** — read target's frontmatter via ruamel.yaml, update only `last_synced`, write back via atomic write. This preserves any user-added frontmatter Glen made on iPhone (Pattern 6 + Pattern 10).

    ```python
    def _update_frontmatter_last_synced(target: Path, last_synced_iso: str) -> None:
        text = target.read_text(encoding="utf-8")
        # Frontmatter delimited by leading '---' line and the next '---' line
        if not text.startswith("---\n"):
            return  # No frontmatter — skip silently (this is unusual but not fatal)
        end_idx = text.find("\n---\n", 4)
        if end_idx == -1:
            return  # Malformed frontmatter — skip
        fm_text = text[4:end_idx]
        body = text[end_idx + 5:]  # after the closing '---\n'

        yaml = _yaml_instance()
        try:
            data = yaml.load(fm_text)
        except Exception:
            return  # YAML parse failed — leave file alone
        if not isinstance(data, dict):
            return
        data["last_synced"] = last_synced_iso

        buf = StringIO()
        yaml.dump(data, buf)
        new_text = f"---\n{buf.getvalue()}---\n{body}"
        # Atomic write
        dir_path = target.parent
        fd, tmp_path = tempfile.mkstemp(dir=dir_path, prefix=f".{target.name}.", suffix=".tmp")
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as f:
                f.write(new_text)
                f.flush()
                os.fsync(f.fileno())
            os.replace(tmp_path, target)
            dir_fd = os.open(dir_path, os.O_RDONLY)
            try:
                os.fsync(dir_fd)
            finally:
                os.close(dir_fd)
        except Exception:
            if os.path.exists(tmp_path):
                try:
                    os.unlink(tmp_path)
                except OSError:
                    pass
            raise
    ```

    ```python
    def run_projection(build_root: Path, icloud_root: Path, dry_run: bool = False) -> dict:
        """D-03 + Pattern 4: project managed sections from vault-build/ to iCloud canonical vault.
        Marker integrity ABORTS per file (D-08a). Returns summary stats."""
        source_dir = build_root / "Clients"
        target_dir = icloud_root / "Clients"
        target_dir.mkdir(parents=True, exist_ok=True)
        last_synced = now_iso_with_offset()

        stats = {"projected": 0, "skipped_marker_error": 0,
                 "skipped_icloud_placeholder": 0, "created_new": 0}
        feed_path = build_root.parent / "data" / "feed.jsonl"  # repo root-relative

        for source in sorted(source_dir.glob("*.md")):
            target = target_dir / source.name

            # Pitfall 2: handle .icloud placeholder
            if not _handle_icloud_placeholder(target):
                stats["skipped_icloud_placeholder"] += 1
                if not dry_run:
                    append_feed_entry(
                        handle_marker_error_for_feed(
                            file_path=target,
                            section="<icloud-placeholder>",
                            reason="brctl download failed after 3 retries",
                        ),
                        feed_path=feed_path,
                    )
                continue

            # Pitfall 6: first-time write — copy full source to target
            if not target.exists():
                if dry_run:
                    print(f"[dry-run] would create new {target}")
                else:
                    _atomic_write(target, source.read_text(encoding="utf-8"))
                stats["created_new"] += 1
                continue

            # Existing target — splice each managed section
            file_aborted = False
            for section in MANAGED_SECTIONS:
                content = _extract_managed_section(source, section)
                if content is None:
                    # Source missing markers — bug in vault_writer's writer side; warn but continue
                    continue
                if dry_run:
                    print(f"[dry-run] would splice {section} into {target}")
                    continue
                try:
                    replace_managed_section(target, section, content)
                except MarkerError as e:
                    # D-08a: ABORT this file's remaining updates and log critical
                    append_feed_entry(
                        handle_marker_error_for_feed(
                            file_path=target,
                            section=section,
                            reason=str(e),
                        ),
                        feed_path=feed_path,
                    )
                    print(f"ABORT marker error: {target} ({section}) — {e}", file=sys.stderr)
                    stats["skipped_marker_error"] += 1
                    file_aborted = True
                    break  # Skip remaining sections for this file

            if file_aborted:
                # Per Pitfall 1: do NOT update last_synced on aborted files;
                # the stale timestamp is the signal Glen sees that something is wrong.
                continue

            if not dry_run:
                _update_frontmatter_last_synced(target, last_synced)
            stats["projected"] += 1

        return stats
    ```

    **File 3: `mac/install-daemon.sh`** — idempotent installer:

    ```bash
    #!/usr/bin/env bash
    # Idempotent installer for the Agend Ops vault-sync LaunchAgent.
    # Usage: bash mac/install-daemon.sh
    #
    # What this does (each step is idempotent):
    #   1. Verifies fswatch is installed; installs via Homebrew if missing.
    #   2. Verifies ruamel.yaml is installed; installs via pip3 if missing.
    #   3. Copies mac/com.agend.vault-sync.plist to ~/Library/LaunchAgents/ if not already in place.
    #   4. Reloads the LaunchAgent (unload-then-load) so changes to the plist or watcher take effect.
    #   5. Verifies the daemon is running.
    #
    # Re-running this script is safe.
    set -euo pipefail

    REPO_ROOT="$(cd "$(dirname "$0")/.." && pwd)"
    PLIST_NAME="com.agend.vault-sync.plist"
    PLIST_SRC="$REPO_ROOT/mac/$PLIST_NAME"
    PLIST_DEST="$HOME/Library/LaunchAgents/$PLIST_NAME"
    LOG_DIR="$HOME/Library/Logs"

    echo "==> Step 1: fswatch"
    if ! command -v fswatch >/dev/null 2>&1; then
      echo "fswatch not found; installing via Homebrew..."
      if ! command -v brew >/dev/null 2>&1; then
        echo "ERROR: Homebrew not found. Install from https://brew.sh and re-run." >&2
        exit 1
      fi
      brew install fswatch
    else
      echo "fswatch present: $(fswatch --version | head -1)"
    fi

    echo "==> Step 2: Python deps"
    if ! python3 -c "import ruamel.yaml" 2>/dev/null; then
      echo "ruamel.yaml not found; installing via pip3..."
      pip3 install -r "$REPO_ROOT/scripts/requirements.txt"
    else
      echo "ruamel.yaml present"
    fi

    echo "==> Step 3: install plist"
    mkdir -p "$LOG_DIR"
    mkdir -p "$(dirname "$PLIST_DEST")"
    if [ ! -f "$PLIST_DEST" ] || ! cmp -s "$PLIST_SRC" "$PLIST_DEST"; then
      cp "$PLIST_SRC" "$PLIST_DEST"
      echo "Installed plist: $PLIST_DEST"
    else
      echo "Plist already up to date: $PLIST_DEST"
    fi

    echo "==> Step 4: ensure watcher script is executable"
    chmod +x "$REPO_ROOT/mac/vault-sync-watcher.sh"

    echo "==> Step 5: reload LaunchAgent (unload-then-load is idempotent)"
    launchctl unload "$PLIST_DEST" 2>/dev/null || true
    launchctl load -w "$PLIST_DEST"

    echo "==> Step 6: verify"
    sleep 1
    if launchctl list | grep -q com.agend.vault-sync; then
      echo "OK: com.agend.vault-sync is loaded."
      echo "Logs: $LOG_DIR/agend-vault-sync.log + agend-vault-sync.err.log"
    else
      echo "WARN: launchctl list does not show com.agend.vault-sync. Check $LOG_DIR/agend-vault-sync.err.log" >&2
      exit 1
    fi

    echo ""
    echo "Daemon installed. Next step: configure the git-pull cadence — see mac/README.md §Cron."
    ```
